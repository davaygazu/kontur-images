function showImage(index) {

    if (!currentProject) {
        return;
    }

    const images =
        currentProject.images;

    if (index < 0) {
        index = images.length - 1;
    }

    if (index >= images.length) {
        index = 0;
    }

    currentIndex = index;
    mainImage.style.objectFit=images[index].includes("-plan-")?"contain":"";

    mainImage.onload=function(){document.getElementById('konturGalleryMain').style.aspectRatio=this.naturalWidth+' / '+this.naturalHeight;};
    mainImage.src =
        previewSource(images[currentIndex]);

    mainImage.alt =
        currentProject.title;

    counter.textContent =
        (currentIndex + 1) +
        " / " +
        images.length;


    const allThumbs =
        thumbs.querySelectorAll(
            ".kontur-gallery__thumb"
        );


    allThumbs.forEach(
        function (thumb, i) {

            thumb.classList.toggle(
                "is-active",
                i === currentIndex
            );

        }
    );


    const activeThumb =
        allThumbs[currentIndex];


    if (activeThumb && modal.classList.contains("is-open")) {

        activeThumb.scrollIntoView({
            behavior: "smooth",
            block: "nearest",
            inline: "center"
        });

    }


    if (
        lightbox.classList.contains(
            "is-open"
        )
    ) {

        updateLightboxImage();

    }

}

        }
    );

}

function createRooms() {

    planButtons.replaceChildren();

            showImage(index);
            openLightbox();
        });
        planButtons.appendChild(button);
    });

    rooms.innerHTML = "";

    currentProject.rooms.forEach(
        function (room) {

            const row =
                document.createElement("div");

            row.className =
                "kontur-room";


            const name =
                document.createElement("span");

            name.className =
                "kontur-room__name";

            name.textContent =
                room[0];


            const value =
                document.createElement("span");

            value.className =
                "kontur-room__value";

            value.textContent =
                room[1];


            row.appendChild(name);
            row.appendChild(value);

            rooms.appendChild(row);

        }
    );

}

    );

function resetZoom() {

    zoom = 1;

    translateX = 0;
    translateY = 0;

    updateTransform();

function updateTransform() {

    lightboxImage.style.transform =
        "translate(" +
        translateX +
        "px, " +
        translateY +
        "px) scale(" +
        zoom +
        ")";


    lightbox.classList.toggle(
        "is-zoomed",
        zoom > 1
    );

}


function updateLightboxImage() {

    zoom += amount;

    if (zoom < 1) {
        zoom = 1;
    }

    if (zoom > 4) {
        zoom = 4;
    }

    if (zoom === 1) {

        translateX = 0;
        translateY = 0;

    }

    updateTransform();

}

function getTouchDistance(touches) {

    const x =
        touches[0].clientX -
        touches[1].clientX;

    const y =
        touches[0].clientY -
        touches[1].clientY;

    return Math.sqrt(
        x * x + y * y
    );

}


lightboxStage.addEventListener(

            event.touches.length === 1

            lightboxTouchStartX =
                event.touches[0].clientX;

                    event.touches

                );

                pinchStartZoom *

                newZoom = 1;
